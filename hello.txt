(async () => {
  const APPLY = false;

  const names = [
    "core_sdk",
    "gui_sdk",
    "plugin_sdk",
    "sura_connector_client",
    "chart_data_manager",
    "charts_common"
  ];

  const api = location.origin + "/app/rest";

  async function req(url, opts) {
    opts = opts || {};
    const headers = Object.assign({ Accept: "application/json" }, opts.headers || {});
    const r = await fetch(url, Object.assign({ credentials: "same-origin" }, opts, { headers: headers }));
    if (!r.ok) throw new Error(r.status + " " + r.statusText + " " + url);
    return r;
  }

  const data = await (await req(api + "/projects")).json();
  const all = data.project || [];

  const plan = [];
  for (const n of names) {
    const hits = all.filter(p => p.name === n);
    let state = "MISSING";
    let id = "";
    if (hits.length === 1) {
      state = "READY";
      id = hits[0].id;
    } else if (hits.length > 1) {
      state = "AMBIGUOUS";
    }
    plan.push({ name: n, to: n.toUpperCase(), id: id, state: state });
  }

  console.table(plan);

  for (const x of plan) {
    if (x.state !== "READY") throw new Error("Есть MISSING или AMBIGUOUS. Ничего не изменено.");
  }

  if (!APPLY) {
    console.log("DRY RUN: TeamCity не изменён. Поставьте APPLY = true.");
    return;
  }

  for (const x of plan) {
    await req(api + "/projects/id:" + encodeURIComponent(x.id) + "/name", {
      method: "PUT",
      headers: { "Content-Type": "text/plain; charset=utf-8" },
      body: x.to
    });
    console.log("RENAMED: " + x.name + " -> " + x.to);
  }

  console.log("ГОТОВО: TeamCity обработан.");
})().catch(e => console.error("ОШИБКА:", e));
