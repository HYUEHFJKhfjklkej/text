#!/usr/bin/env python3
"""Прописывает NuGet.Config для сборки Elara-проектов на legacy CMake-фреймворке.

Фреймворк читает NuGet.Config сам, через cmake/NuGetProto.cmake. Без этого файла
configure падает на `NuGet config file "..." doesn't exists.`

Запуск:
    python setup_nuget_config.py              прописать файл и проверить фиды
    python setup_nuget_config.py --check      только проверить, ничего не менять
    python setup_nuget_config.py --insecure   не проверять TLS-сертификат ProGet
    python setup_nuget_config.py --print      показать путь к файлу и выйти

Работает и на Windows, и на Linux: путь к NuGet.Config выбирается по платформе.
Существующий файл не затирается молча, рядом кладётся копия с меткой времени.
"""

import argparse
import os
import re
import shutil
import ssl
import sys
import time
import urllib.error
import urllib.request

PROGET = "https://proget.inc.elara.local"

# Порядок важен: резолвер перебирает фиды сверху в
FEEDS = [
    ("Default", PROGET + "/nuget/Default/"),
    ("Prerelease", PROGET + "/nuget/Prerelease/"),
    ("Legacy", PROGET + "/nuget/Legacy/"),
    ("External", PROGET + "/nuget/External/"),
    ("Conan", PROGET + "/nuget/Conan/"),
]

# Пакеты, которые ищем в фиде Conan для проверки. Слот win.v142.shared.x64
# соответствует BUILD_SHARED_LIBS=ON (дефолт легас.
PROBE_PACKAGES = [
    "rapidjson.win.v142.shared.x64",
    "rapidjson.win.v142.static.x64",
]

TIMEOUT = 15


def config_path():
    if os.name == "nt":
        base = os.environ.get("APPDATA")
        if not base:
            sys.exit("Не задана переменная APPDATA, не могу определить путь к NuGet.Config")
        return os.path.join(base, "NuGet", "NuGet.
    return os.path.join(os.path.expanduser("~"), ".config", "NuGet", "NuGet.Config")


def render():
    lines = [
        '<?xml version="1.0" encoding="utf-8"?>',
        "<configuration>",
        "  <packageSources>",
        "    <clear />",
    ]
    for key, url in FEEDS:
        lines.append('    <add key="%s" value="%s"
    lines += ["  </packageSources>", "</configuration>", ""]
    return "\n".join(lines)


def write_config(path):
    content = render()

    if os.path.exists(path):
        with open(path, "r", encoding="utf-8-sig") as handle:
            if handle.read() == content:
                print("[=] Файл уже в нужном виде: %s" % path)
                return False
        backup = "%s.%s.bak" % (path, time.strftime("%Y%m%d-%H%M%S"))
        shutil.copy2(path, backup)
        print("[i] Старый файл сохранён: %s" % backup)

    os.makedirs(os.path.dirname(path), exist_ok=True)
    # utf-8 без BOM: некоторые парсеры NuGet на BO
    with open(path, "w", encoding="utf-8", newline="\n") as handle:
        handle.write(content)
    print("[+] Записан %s" % path)
    return True


def opener(insecure):
    if not insecure:
        return urllib.request.build_opener()
    context = ssl.create_default_context()
    context.check_hostname = False
    context.verify_mode = ssl.CERT_NONE
    return urllib.request.build_opener(urllib.request.HTTPSHandler(context=context))

def check_packages(client):
    feed = dict(FEEDS)["Conan"]
    for package in PROBE_PACKAGES:
        url = "%sFindPackagesById()?id='%s'" % (feed, package)
        try:
            _, body = fetch(client, url)
        except Exception as error:
            print("[!] %-34s запрос не прошёл: %s" % (package, error))
            continue
        versions = re.findall(r"<d:Version>([^<]+)</d:Version>", body)
        if versions:
            print("[+] %-34s %s" % (package, ", ".join(sorted(set(versions)))))
        else:
            print("[!] %-34s в фиде Conan не найден" % package)

def main():
    parser = argparse.ArgumentParser(description=" legacy CMake-фреймворка")
    parser.add_argument("--check", action="store_true", help="только проверить, файл не трогать")
    parser.add_argument("--insecure", action="storь TLS-сертификат ProGet")
    parser.add_argument("--print", dest="show", action="store_true", help="показать путь и выйти")
    args = parser.parse_args()

    path = config_path()

    if args.show:
        print(path)
        return 0
